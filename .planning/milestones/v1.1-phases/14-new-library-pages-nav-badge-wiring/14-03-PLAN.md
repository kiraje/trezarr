    - DropdownMenu, DropdownMenuTrigger, DropdownMenuContent, DropdownMenuItem from
      "../components/ui/dropdown-menu"
    - ChevronUp, ChevronDown from "lucide-react" (for sort direction indicators)

    STATE:
    - data: LibraryResponse | null
    - loading: boolean (initial true)
    - error: string | null
    - searchInput: string (raw input, not yet debounced)
    - query: string (debounced — what filtering uses)
    - filterStatus: "all" | "needs_vi" | "translated" | "no_source"
    - sortField: "title" | "progress"
    - sortDir: "asc" | "desc"
    - translateError: never used in this page (Series list has no Translate button)

    CONTEXT:
    const { setLibraryData } = useLibraryContext();

    FETCH (useEffect, []):
      async function load() {
        setLoading(true); setError(null);
        try {
          const result = await getLibrary();
          setData(result);
          setLibraryData(result);   // D-09: populate context for sidebar badges
        } catch (err) {
          setError(String(err));
        } finally {
          setLoading(false);
        }
      }
      void load(); — wrap in async IIFE, not direct async to useEffect

    DEBOUNCE (useEffect, [searchInput]):
      const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
      useEffect(() => {
        if (timerRef.current) clearTimeout(timerRef.current);
        timerRef.current = setTimeout(() => setQuery(searchInput), 250);
        return () => { if (timerRef.current) clearTimeout(timerRef.current); };
      }, [searchInput]);

    DERIVED DATA (computed inline before return, not in state):
      Derive isSonarrDown = data !== null && data.series.length === 0 &&
        data.errors.some(e => e.source === "sonarr");

      Filter series:
        let filtered = data?.series ?? [];
        if (query) filtered = filtered.filter(s =>
          s.title.toLowerCase().includes(query.toLowerCase()));
        if (filterStatus === "needs_vi") filtered = filtered.filter(s =>
          s.translated_count < s.total_count);
        if (filterStatus === "translated") filtered = filtered.filter(s =>
          s.translated_count >= s.total_count && s.total_count > 0);
        if (filterStatus === "no_source") filtered = filtered.filter(s =>
          s.total_count === 0);

      Sort series:
        if (sortField === "title") {
          filtered = [...filtered].sort((a, b) =>
            sortDir === "asc" ? a.title.localeCompare(b.title) :
              b.title.localeCompare(a.title));
        } else {
          filtered = [...filtered].sort((a, b) => {
            const ra = a.total_count > 0 ? a.translated_count / a.total_count : 0;
            const rb = b.total_count > 0 ? b.translated_count / b.total_count : 0;
            return sortDir === "asc" ? ra - rb : rb - ra;
          });
        }

    RENDER STRUCTURE per 14-UI-SPEC.md §Series List Page:

    LOADING STATE (when loading === true):
      <div className="space-y-4">
        <Skeleton className="h-8 w-32 mb-4" />
        <Skeleton className="h-9 w-64 mb-4" />
        {Array.from({length: 6}).map((_, i) => (
          <Skeleton key={i} className="h-12 w-full mb-1" />

      </div>

    ERROR STATE (when error !== null, i.e. network/fetch threw):
      Heading "Failed to load Series" + body "An error occurred while fetching the library.
      Try reloading." + <Button onClick={() => void load()}>Retry</Button> (default variant)
      — store the load function in a ref or extract it so retry works.

    SONARR DOWN STATE (when isSonarrDown):
      Heading "Sonarr unavailable" + body "Could not load the series library. Check your Sonarr
      connection in Settings." + <Button variant="outline" onClick={() => navigate('/settings')}>
      Go to Settings</Button>

    NORMAL STATE:
      <div className="space-y-4">
        <h1 className="text-xl font-semibold">Series</h1>

        {/* Controls bar */}
        <div className="flex items-center gap-3 mb-4">
          <Input
            placeholder="Search series..."
            className="max-w-xs"
            value={searchInput}
            onChange={e => setSearchInput(e.target.value)}
          />
          <DropdownMenu>
            <DropdownMenuTrigger asChild>
              <Button variant="outline" size="sm">
                Filter: {filterLabel}
              </Button>
            </DropdownMenuTrigger>
            <DropdownMenuContent>
              <DropdownMenuItem onSelect={() => setFilterStatus("all")}>All</DropdownMenuItem>
              <DropdownMenuItem onSelect={() => setFilterStatus("needs_vi")}>Needs VI</DropdownMenuItem>
              <DropdownMenuItem onSelect={() => setFilterStatus("translated")}>Translated</DropdownMenuItem>
              <DropdownMenuItem onSelect={() => setFilterStatus("no_source")}>No source</DropdownMenuItem>
            </DropdownMenuContent>
          </DropdownMenu>
          <Button variant="outline" size="sm"
            onClick={() => {
              if (sortField === "title") {
                setSortDir(d => d === "asc" ? "desc" : "asc");
              } else {
                setSortField("title"); setSortDir("asc");
              }
            }}
          >
            Sort by Title {sortField === "title" ? (sortDir === "asc" ? "↑" : "↓") : ""}
          </Button>
          <Button variant="outline" size="sm"
            onClick={() => {
              if (sortField === "progress") {
                setSortDir(d => d === "asc" ? "desc" : "asc");
              } else {
                setSortField("progress"); setSortDir("desc");
              }
            }}
          >
            Sort by Progress {sortField === "progress" ? (sortDir === "asc" ? "↑" : "↓") : ""}
          </Button>
        </div>

        EMPTY AFTER FILTER (filtered.length === 0 && !isSonarrDown):
          Heading "No series found" + body "No series match your search or filter. Try clearing the
          filter." + <Button variant="outline" onClick={() => { setSearchInput(""); setQuery("");
          setFilterStatus("all"); }}>Clear filters</Button>

        TABLE (filtered.length > 0):
          Plain HTML table — the installed table.tsx is jolly-ui React Aria; do NOT import it.
          Use a raw <table> element directly in this page (D-01: plain HTML table for sort stability;
          jolly-ui Table beta risk avoided as prescribed in 14-UI-SPEC.md D-01 re-vet note).

          <div className="overflow-x-auto">
            <table className="w-full text-sm">
              <thead>
                <tr className="border-b border-border text-muted-foreground">
                  <th className="text-left py-2 px-4">Title</th>
                  <th className="text-left py-2 px-4 w-[80px]">Year</th>
                  <th className="text-left py-2 px-4 w-[220px]">Progress</th>

                </tr>

                {filtered.map(s => (

                  >
                    <td className="py-2 px-4">

                      {s.year && <span className="text-muted-foreground text-xs ml-2">({s.year})</span>}
                    </td>

                      {ProgressBar for s — see below}

                      {s.total_count === 0 ? '—' : `${s.translated_count}/${s.total_count}`}
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>

    PROGRESS BAR (inline helper function ProgressBar({ item: SeriesItem }) or inline JSX):
      When item.total_count === 0: render <span className="text-muted-foreground">—</span>

        no_source: "No source" }[filterStatus];
