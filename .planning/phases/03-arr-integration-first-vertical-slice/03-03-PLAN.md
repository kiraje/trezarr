      """Trezarr *arr discovery layer — Sonarr + Radarr API wrappers.

      Module-level helpers:
        DiscoveryError      — raised by discover_sonarr_items / discover_radarr_items when the
                              underlying httpx layer fails. cli.py catches this per-service to
                              implement D-30-style per-service resilience.
        _normalize_arr_host — accepts bare hostnames or full URLs and returns the bare host.
      """
      from __future__ import annotations
      from urllib.parse import urlparse

      __all__ = ["DiscoveryError", "_normalize_arr_host"]


      class DiscoveryError(Exception):
          """Raised when an *arr discovery call fails at the HTTP transport layer.

          Wraps httpx.HTTPStatusError AND httpx.RequestError so callers (cli.py)
          can apply per-service resilience without depending on httpx directly.
          """


      def _normalize_arr_host(host: str) -> str:
          """Accept either a bare hostname/IP or a full URL; return the bare host.

          Examples:
              "192.168.1.10"                  -> "192.168.1.10"
              "http://192.168.1.10:8989"      -> "192.168.1.10"
              "https://sonarr.example/api"    -> "sonarr.example"
              "sonarr.lan"                    -> "sonarr.lan"

          The bare host returned here is passed to `pyarr.Sonarr(host=..., port=...)`
          which itself reconstructs the URL using the separately-configured port.
          """
          # If no scheme present, return as-is (the bare hostname/IP case).
          if "://" not in host:
              return host
          parsed = urlparse(host)
          # parsed.hostname strips port; falls back to the netloc if hostname is None.
          return parsed.hostname or parsed.netloc.split(":")[0]

          try:
              client = build_sonarr_client(settings)
              series_list = client.series.get()
              items: list[MediaItem] = []
              for series in series_list:
                  if not series.get("monitored", False):
                      continue
                  ep_files = client.episode_file.get(series_id=series["id"])
                  for ep_file in ep_files:
                      raw_path = ep_file.get("path")
                      if raw_path is None:
                          logger.warning("Sonarr ep_file %s has no path; skipping", ep_file.get("id"))
                          continue
                      local_path = apply_path_mapping(raw_path, settings.path_mappings)

                          title=series["title"],
                          source_type="episode",
                          series_id=series["id"],
                          season_number=ep_file.get("seasonNumber"),
                      ))
              return items
          except httpx.HTTPStatusError as exc:
              logger.error("Sonarr HTTP %s at %s:%d — %s",

                  f"Sonarr discovery failed: HTTP {exc.response.status_code} "
                  f"at {settings.sonarr_host}:{settings.sonarr_port}"

          except httpx.RequestError as exc:
              logger.error("Sonarr transport error at %s:%d — %s",

                  f"Sonarr discovery failed: transport error at "
                  f"{settings.sonarr_host}:{settings.sonarr_port}: {exc}"
              ) from exc

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Sonarr/Radarr API response → MediaItem | API-returned paths are untrusted strings; path mapping is applied before they enter MediaItem.local_path |
| TrezarrSettings → pyarr client | SecretStr resolved at client construction only; never stored on client or in MediaItem |
| httpx transport layer → cli.py | DiscoveryError is the typed boundary; raw httpx exceptions never escape the discovery module |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-03-01 | Information Disclosure | sonarr_api_key / radarr_api_key in logs or MediaItem | mitigate | SecretStr; .get_secret_value() called only inside build_sonarr_client() / build_radarr_client(); MediaItem contains no key fields; logger calls never log settings directly |
| T-03-03-02 | Tampering | Arbitrary path injection via *arr API response (movie['path'] injection) | mitigate | apply_path_mapping applied before any filesystem use; assert_within_media_roots() in Plan 03-04 / cli.py validates every resolved write path; Pitfall 2 guard ensures movie directory vs. video file distinction; defensive .get("path") check on every API response |
| T-03-03-03 | Elevation of Privilege | pyarr connecting to a Sonarr/Radarr running as root with write access | accept | Trezarr only reads from Sonarr/Radarr (GET endpoints); no write calls to the *arr API in Phase 3; discovery is read-only |
| T-03-03-04 | Denial of Service | httpx.HTTPStatusError or httpx.RequestError abort the full run | mitigate | Caught at the discovery boundary and re-raised as DiscoveryError with host/port context; cli.py (Plan 03-05) catches DiscoveryError per-service so one bad *arr does NOT kill the run if the other *arr is healthy (per 03-REVIEWS.md MEDIUM #10) |
</threat_model>
