        setTranslateErrors(prev => { const next = {...prev}; delete next[movie.id]; return next; });
        try {
          // source_path is not available — this will 400 from the backend ("source_path_required").
          // The button is disabled unless source_path is available, so this branch only fires if

        } finally {
          setTranslatePending(null);
        }
      }
